     4→// ============================================================
     5→// Type Definitions

     7→
     8→interface SongData {
     9→  id: string
    10→  title: string
    11→  artist: string
    12→  album?: string
    13→  duration: number
    14→  coverUrl?: string
    15→  audioUrl?: string
    16→}
    17→
    18→interface MemberState {
    19→  socketId: string
    20→  userId: string
    21→  userName: string
    22→  userImage?: string
    23→  joinedAt: number
    24→  isMuted: boolean
    25→}
    26→
    27→interface PlaybackState {
    28→  isPlaying: boolean
    29→  currentSong: SongData | null
    30→  position: number // in milliseconds
    31→  lastSyncTime: number // timestamp of last sync
    32→  startedAt: number // when playback started (for sync calculation)
    33→}
    34→
    35→interface QueueItem {
    36→  song: SongData
    37→  addedBy: string
    38→  addedAt: number
    39→}
    40→
    41→interface RoomState {
    42→  id: string
    43→  name: string
    44→  hostId: string
    45→  members: Map<string, MemberState>
    46→  playback: PlaybackState
    47→  queue: QueueItem[]
    48→  voiceMembers: Set<string>
    49→}
    50→
    51→// ============================================================
    52→// Room & Socket Tracking
    53→// ============================================================
    54→
    55→const rooms = new Map<string, RoomState>()
    56→const socketToRoom = new Map<string, string>() // socketId -> roomId
    57→const socketToUser = new Map<string, string>() // socketId -> userId
    58→
    59→function getOrCreateRoom(roomId: string, roomName: string, hostId: string): RoomState {
    60→  if (!rooms.has(roomId)) {
    61→    rooms.set(roomId, {
    62→      id: roomId,
    63→      name: roomName,
    64→      hostId,
    65→      members: new Map(),
    66→      playback: {
    67→        isPlaying: false,
    68→        currentSong: null,
    69→        position: 0,
    70→        lastSyncTime: 0,
    71→        startedAt: 0,
    72→      },
    73→      queue: [],
    74→      voiceMembers: new Set(),
    75→    })
    76→  }
    77→  return rooms.get(roomId)!
    78→}
    79→
    80→function getRoomForSocket(socket: Socket): RoomState | null {
    81→  const roomId = socketToRoom.get(socket.id)
    82→  if (!roomId) return null
    83→  return rooms.get(roomId) ?? null
    84→}
    85→
    86→function removeMemberFromRoom(socket: Socket): { roomId: string; userId: string; userName: string } | null {
    87→  const roomId = socketToRoom.get(socket.id)
    88→  const userId = socketToUser.get(socket.id)
    89→  if (!roomId || !userId) return null
    90→
    91→  const room = rooms.get(roomId)
    92→  if (!room) return null
    93→
    94→  const member = room.members.get(userId)
    95→  const userName = member?.userName ?? 'Unknown'
    96→
    97→  room.members.delete(userId)
    98→  room.voiceMembers.delete(userId)
    99→  socketToRoom.delete(socket.id)
   100→  socketToUser.delete(socket.id)
   101→
   102→  // If room is empty, clean it up
   103→  if (room.members.size === 0) {
   104→    rooms.delete(roomId)
   105→    console.log(`Room ${roomId} deleted (empty)`)
   106→  } else {
   107→    // If the host left, assign a new host (earliest member)
   108→    if (room.hostId === userId) {
   109→      const earliestMember = Array.from(room.members.values()).sort(
   110→        (a, b) => a.joinedAt - b.joinedAt
   111→      )[0]
   112→      if (earliestMember) {
   113→        room.hostId = earliestMember.userId
   114→        // Notify room about host change
   115→        io.to(roomId).emit('room:host-change', {
   116→          roomId,
   117→          newHostId: earliestMember.userId,
   118→          newHostName: earliestMember.userName,
   119→        })
   120→        console.log(`Host transferred to ${earliestMember.userName} in room ${roomId}`)
   121→      }
   122→    }
   123→  }
   124→
   125→  return { roomId, userId, userName }
   126→}
   127→
   128→function serializeRoom(room: RoomState) {
   129→  return {
   130→    id: room.id,
   131→    name: room.name,
   132→    hostId: room.hostId,
   133→    members: Array.from(room.members.values()).map((m) => ({
   134→      userId: m.userId,
   135→      userName: m.userName,
   136→      userImage: m.userImage,
   137→      joinedAt: m.joinedAt,
   138→      isMuted: m.isMuted,
   139→    })),
   140→    playback: {
   141→      isPlaying: room.playback.isPlaying,
   142→      currentSong: room.playback.currentSong,
   143→      position: room.playback.position,
   144→      lastSyncTime: room.playback.lastSyncTime,
   145→      startedAt: room.playback.startedAt,
   146→    },
   147→    queue: room.queue,
   148→    voiceMembers: Array.from(room.voiceMembers),
   149→  }
   150→}
   151→
   152→function calculateCurrentPosition(room: RoomState): number {
   153→  if (!room.playback.isPlaying || room.playback.startedAt === 0) {
   154→    return room.playback.position
   155→  }
   156→  const elapsed = Date.now() - room.playback.startedAt
   157→  return room.playback.position + elapsed
   158→}

   187→  socket.on('room:join', (data: { roomId: string; roomName?: string; userId: string; userName: string; userImage?: string }) => {
   188→    try {
   189→      const { roomId, roomName, userId, userName, userImage } = data
   190→
   191→      if (!roomId || !userId || !userName) {
   192→        socket.emit('error', { message: 'Missing required fields: roomId, userId, userName' })
   193→        return
   194→      }
   195→
   196→      // Leave previous room if in one
   197→      const prevRoomId = socketToRoom.get(socket.id)
   198→      if (prevRoomId) {
   199→        const leaveInfo = removeMemberFromRoom(socket)
   200→        if (leaveInfo) {
   201→          socket.leave(prevRoomId)
   202→          io.to(prevRoomId).emit('room:member-left', {
   203→            roomId: prevRoomId,
   204→            userId: leaveInfo.userId,
   205→            userName: leaveInfo.userName,
   206→          })
   207→          const prevRoom = rooms.get(prevRoomId)
   208→          if (prevRoom) {
   209→            io.to(prevRoomId).emit('room:update', serializeRoom(prevRoom))
   210→          }
   211→        }
   212→      }
   213→
   214→      // Get or create room
   215→      const room = getOrCreateRoom(roomId, roomName ?? `Room ${roomId}`, userId)
   216→
   217→      // Add member to room
   218→      const member: MemberState = {
   219→        socketId: socket.id,
   220→        userId,
   221→        userName,
   222→        userImage,
   223→        joinedAt: Date.now(),
   224→        isMuted: false,
   225→      }
   226→      room.members.set(userId, member)
   227→
   228→      // Track socket
   229→      socketToRoom.set(socket.id, roomId)
   230→      socketToUser.set(socket.id, userId)
   231→
   232→      // Join socket.io room
   233→      socket.join(roomId)
   234→
   235→      // Send current room state to the joining user
   236→      socket.emit('room:joined', {
   237→        room: serializeRoom(room),
   238→        currentPosition: calculateCurrentPosition(room),
   239→      })
   240→
   241→      // Notify others in the room
   242→      socket.to(roomId).emit('room:member-joined', {
   243→        roomId,
   244→        member: {
   245→          userId,
   246→          userName,
   247→          userImage,
   248→          joinedAt: member.joinedAt,
   249→          isMuted: member.isMuted,
   250→        },
   251→      })
   252→
   253→      // Broadcast updated room state
   254→      io.to(roomId).emit('room:update', serializeRoom(room))
   255→
   256→      console.log(`[ROOM:JOIN] ${userName} (${userId}) joined room ${roomId}. Members: ${room.members.size}`)
   257→    } catch (error) {
   258→      console.error('[ROOM:JOIN] Error:', error)
   259→      socket.emit('error', { message: 'Failed to join room' })
   260→    }
   261→  })
   262→
   263→  socket.on('room:leave', (data: { roomId: string; userId: string }) => {
   264→    try {
   265→      const { roomId, userId } = data
   266→      const currentRoomId = socketToRoom.get(socket.id)
   267→
   268→      if (currentRoomId !== roomId) {
   269→        socket.emit('error', { message: 'You are not in this room' })
   270→        return
   271→      }
   272→
   273→      socket.leave(roomId)
   274→      const leaveInfo = removeMemberFromRoom(socket)
   275→
   276→      if (leaveInfo) {
   277→        io.to(roomId).emit('room:member-left', {
   278→          roomId,
   279→          userId: leaveInfo.userId,
   280→          userName: leaveInfo.userName,
   281→        })
   282→        const room = rooms.get(roomId)
   283→        if (room) {
   284→          io.to(roomId).emit('room:update', serializeRoom(room))
   285→        }
   286→      }
   287→
   288→      socket.emit('room:left', { roomId })
   289→      console.log(`[ROOM:LEAVE] ${userId} left room ${roomId}`)
   290→    } catch (error) {
   291→      console.error('[ROOM:LEAVE] Error:', error)
   292→      socket.emit('error', { message: 'Failed to leave room' })
   293→    }
   294→  })

   300→  socket.on('music:play', (data: { roomId: string; songId?: string; position?: number; timestamp: number }) => {
   301→    try {
   302→      const { roomId, position, timestamp } = data
   303→      const room = rooms.get(roomId)
   304→      if (!room) {
   305→        socket.emit('error', { message: 'Room not found' })
   306→        return
   307→      }
   308→
   309→      // Only host can control playback (or allow any member for flexibility)
   310→      const userId = socketToUser.get(socket.id)
   311→      const member = userId ? room.members.get(userId) : null
   312→      if (!member) {
   313→        socket.emit('error', { message: 'You are not a member of this room' })
   314→        return
   315→      }
   316→
   317→      const currentPosition = position ?? room.playback.position
   318→      room.playback.isPlaying = true
   319→      room.playback.position = currentPosition
   320→      room.playback.startedAt = Date.now()
   321→      room.playback.lastSyncTime = timestamp
   322→
   323→      // Broadcast to all other room members
   324→      socket.to(roomId).emit('music:play', {
   325→        roomId,
   326→        songId: room.playback.currentSong?.id,
   327→        position: currentPosition,
   328→        timestamp,
   329→        startedAt: room.playback.startedAt,
   330→      })
   331→
   332→      console.log(`[MUSIC:PLAY] Room ${roomId} - Position: ${currentPosition}ms`)
   333→    } catch (error) {
   334→      console.error('[MUSIC:PLAY] Error:', error)
   335→      socket.emit('error', { message: 'Failed to play music' })
   336→    }
   337→  })

   451→  socket.on('music:sync', (data: { roomId: string; position: number; timestamp: number; isPlaying: boolean }) => {
   452→    try {
   453→      const { roomId, position, timestamp, isPlaying } = data

   455→      if (!room) {
   456→        socket.emit('error', { message: 'Room not found' })

   468→        room.playback.startedAt = 0
   469→      }
   470→
   471→      // Broadcast sync to other room members
   472→      socket.to(roomId).emit('music:sync', {
   473→        roomId,
   474→        position,
   475→        timestamp,
   476→        isPlaying,
   477→        startedAt: room.playback.startedAt,
   478→      })
   479→
   480→      console.log(`[MUSIC:SYNC] Room ${roomId} - Position: ${position}ms, Playing: ${isPlaying}`)

   482→      console.error('[MUSIC:SYNC] Error:', error)
   483→      socket.emit('error', { message: 'Failed to sync music' })
   484→    }

   552→  socket.on('queue:remove', (data: { roomId: string; songId: string }) => {
   553→    try {
   554→      const { roomId, songId } = data
   555→      const room = rooms.get(roomId)
   556→      if (!room) {
   557→        socket.emit('error', { message: 'Room not found' })
   558→        return
   559→      }
   560→
   561→      const index = room.queue.findIndex((item) => item.song.id === songId)
   562→      if (index === -1) {
   563→        socket.emit('error', { message: 'Song not found in queue' })
   564→        return
   565→      }
   566→
   567→      const removed = room.queue.splice(index, 1)[0]
   568→
   569→      io.to(roomId).emit('queue:update', {
   570→        roomId,
   571→        queue: room.queue,
   572→        action: 'remove',
   573→        songId,
   574→      })
   575→
   576→      console.log(`[QUEUE:REMOVE] Room ${roomId} - Removed "${removed.song.title}"`)
   577→    } catch (error) {
   578→      console.error('[QUEUE:REMOVE] Error:', error)
   579→      socket.emit('error', { message: 'Failed to remove song from queue' })
   580→    }
   581→  })

   624→        roomId,

   631→        roomId,

   904→    }
   905→
   906→    // Delete empty rooms

   908→      rooms.delete(roomId)
